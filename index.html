Net.Event("Shoot").OnServerEvent:Connect(function(player, hitPart, muzzlePos, endPos, toolName)
	local settings = GunSettings.Guns[toolName]
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local tool = character and character:FindFirstChild(toolName)
	if not settings or not root or not tool then return end
	if not allowShot(player, settings.FireRate) then return end

	local reach = MUZZLE_REACH + root.AssemblyLinearVelocity.Magnitude * LAG_TIME
	if (muzzlePos - root.Position).Magnitude > reach then return end
	if (endPos - muzzlePos).Magnitude > settings.Range + RANGE_SLACK then return end

	if hitPart then
		local slack = hitPart.Size.Magnitude / 2 + HIT_SLACK + hitPart.AssemblyLinearVelocity.Magnitude * LAG_TIME
		if distanceToSegment(hitPart.Position, muzzlePos, endPos) > slack then
			hitPart = nil
		end
	end

	GunService.Shoot(player, hitPart, muzzlePos, endPos, toolName)
end)
